FORM check_input.
  RANGES:r_werks FOR t001w-werks.

* 工廠範圍
  REFRESH:r_werks.
  SELECT * FROM t001k
  WHERE bukrs IN s_bukrs
  AND bwkey IN s_plant.
    r_werks-sign = 'I'.
    r_werks-option = 'EQ'.
    r_werks-low = t001k-bwkey.
    APPEND r_werks. CLEAR r_werks.
  ENDSELECT.
  IF sy-subrc <> 0.
    MESSAGE i004(zmm) WITH
    'No plant found according to the entered conditions'(001).
    STOP.
  ENDIF.
*抓取物料群組說明
  REFRESH:itab_017.
  SELECT * FROM zidpmm017 INTO CORRESPONDING FIELDS OF TABLE itab_017
    WHERE werks IN r_werks.

* 倉碼範圍
  REFRESH:itab_016.
  IF p_nco = ''.
    SELECT * FROM zidpmm016 INTO CORRESPONDING FIELDS OF TABLE itab_016
    WHERE werks IN r_werks
    AND lgort IN s_sloc
    AND indicator1 = 'Y'.
*    AND active = 'Y'.  "只考慮有效倉碼，減少無用數據
  ELSE.
    SELECT * FROM zidpmm016 INTO CORRESPONDING FIELDS OF TABLE itab_016
    WHERE werks IN r_werks
    AND lgort IN s_sloc.
*    AND active = 'Y'.
  ENDIF.
  IF itab_016[] IS INITIAL.
    MESSAGE i004(zmm) WITH
    'No location found according to the entered conditions'(002).
    STOP.
  ENDIF.

  SELECT * INTO CORRESPONDING FIELDS OF TABLE itab_023
  FROM t023t.


* 天數區間
  IF t_day1 IS INITIAL.
    EXIT.
  ENDIF.

  itab_date-serial = 1.
  itab_date-date = t_day1.
  APPEND itab_date.
  CLEAR itab_date.

  IF NOT t_day2 IS INITIAL.
    IF t_day2 < t_day1.
      EXIT.
    ENDIF.
    itab_date-serial = 2.
    itab_date-date = t_day2.
    APPEND itab_date.
    CLEAR itab_date.
  ENDIF.

  IF NOT t_day3 IS INITIAL.
    IF t_day3 < t_day2.
      EXIT.
    ENDIF.
    itab_date-serial = 3.
    itab_date-date = t_day3.
    APPEND itab_date.
    CLEAR itab_date.
  ENDIF.

  IF NOT t_day4 IS INITIAL.
    IF t_day4 < t_day3.
      EXIT.
    ENDIF.
    itab_date-serial = 4.
    itab_date-date = t_day4.
    APPEND itab_date.
    CLEAR itab_date.
  ENDIF.

  IF NOT t_day5 IS INITIAL.
    IF t_day5 < t_day4.
      EXIT.
    ENDIF.
    itab_date-serial = 5.
    itab_date-date = t_day5.
    APPEND itab_date.
    CLEAR itab_date.
  ENDIF.

*---------------------------------------------------------------------*
*       FORM task_launch                                              *
*---------------------------------------------------------------------*
*       ........                                                      *
*---------------------------------------------------------------------*
*  -->  P_PARA_IMP                                                    *
*  -->  P_PARA_EXP                                                    *
*  -->  P_FAILED_OBJS                                                 *
*  -->  P_PENDING_OBJS                                                *
*  -->  P_USER_PARAM                                                  *
*---------------------------------------------------------------------*
FORM task_launch
   USING
      p_para_imp   TYPE spta_t_para_imp
   CHANGING
      p_para_exp   TYPE spta_t_para_exp
      p_failed_objs     TYPE spta_t_failed_objects
      p_pending_objs    TYPE spta_t_pending_objects
      p_user_param      TYPE t_user_param.

  DATA:l_package_size LIKE sy-tabix,
       l_failed_obj   LIKE LINE OF p_failed_objs,
       l_pending_obj  LIKE LINE OF p_pending_objs.
  DATA:lt_objects LIKE zidpmmaging OCCURS 0 WITH HEADER LINE.
  DATA:l_nr_obj_left   LIKE sy-index,
       l_package_start TYPE i,              " From-Index des Pakets
       l_package_end   TYPE i,              "   To-Index des Pakets
       l_tabix         LIKE sy-tabix,
       ln_pend_obj     TYPE LINE OF kkpa_t_pending_objects,
       ln_fail_obj     TYPE LINE OF kkpa_t_failed_objects,
       ln_objdata      LIKE zidpmmaging.
  CONSTANTS:p_min TYPE i VALUE 10,
            p_max TYPE i VALUE 30.
  DATA:l_level(1)  TYPE c,
       l_baseon(1) TYPE c,
       l_sttype(1) TYPE c.

  IF p_failed_objs[] IS INITIAL.
    l_nr_obj_left = p_user_param-to_do - g_obj_done.
    IF l_nr_obj_left IS INITIAL.
      p_para_exp-all_submit = 'X'.
    ELSE.
      l_package_size = l_nr_obj_left / p_para_imp-nr_total_tasks.
      IF l_package_size < p_min.
        l_package_size = p_min.
      ELSEIF l_package_size > p_max.
        l_package_size = p_max.
      ENDIF.
      l_package_start = g_obj_done + 1.              "From-Index
      l_package_end   = g_obj_done + l_package_size. "To-Index
      CLEAR ln_pend_obj.
      CLEAR ln_objdata.
      LOOP AT p_user_param-objects INTO ln_objdata FROM l_package_start
                                                   TO l_package_end.
        APPEND ln_objdata TO lt_objects.
        ADD 1 TO g_obj_done.
        ln_pend_obj-obj_id = g_obj_done.    "entspricht Index in Tab.
        APPEND ln_pend_obj TO p_pending_objs.
      ENDLOOP.
    ENDIF.
  ELSE.
    READ TABLE p_failed_objs INTO ln_fail_obj INDEX 1.
    APPEND ln_fail_obj TO p_pending_objs.
    l_tabix = ln_fail_obj-obj_id.
    READ TABLE p_user_param-objects INDEX l_tabix INTO ln_objdata.
    APPEND ln_objdata TO lt_objects.
    DELETE p_failed_objs INDEX 1.
  ENDIF.

  IF lt_objects[] IS INITIAL.
    p_para_exp-rfc_subrc  = kkpa_rfc_subrc-rfc_not_started.
    p_para_exp-all_submit = 'X'.
    EXIT.
  ENDIF.
* 庫齡層級
  IF p_bu = 'X'.
    l_level = '3'.
  ELSEIF p_pla = 'X'.
    l_level = '2'.
  ELSEIF p_loc = 'X'.
    l_level = '1'.
  ELSEIF p_wl = 'X'.
    l_level = '4'.
  ENDIF.
* 以過帳/開單日
  IF p_cd = 'X'.
    l_baseon = '2'.
  ELSE.
    l_baseon = '1'.
  ENDIF.
* 考慮非成本
  IF p_nco = 'X'.
    l_sttype = '1'.
  ELSE.
    l_sttype = '2'.
  ENDIF.

  CALL FUNCTION 'ZRFC_MM_IDP_0294'
  STARTING NEW TASK p_para_imp-taskname
       DESTINATION p_para_imp-taskdest
       PERFORMING (p_para_imp-eot_form) ON END OF TASK
       EXPORTING
            i_level = l_level
            i_basedate = p_date
            i_baseon = l_baseon
            i_stdate = g_date
            i_sttime = g_time
            i_sttype = l_sttype
            i_age07 = p_age07
            i_usecache = p_usec
       TABLES
            it_date = itab_date
            it_data = lt_objects
            it_mov = s_mov
            it_lgort = s_sloc
      EXCEPTIONS
         system_failure        = 1
         communication_failure = 2
         OTHERS                = 3.

ENDFORM.

*---------------------------------------------------------------------*
*       FORM get_task_results                                         *
*---------------------------------------------------------------------*
*       ........                                                      *
*---------------------------------------------------------------------*
*  -->  P_SPTA_PARA_IMP                                               *
*  -->  P_PENDING_OBJS                                                *
*  -->  P_SPTA_PARA_EXP                                               *
*  -->  P_USER_PARAM                                                  *
*---------------------------------------------------------------------*
FORM get_task_results
   USING
      p_spta_para_imp   TYPE spta_t_para_imp
      p_pending_objs    TYPE spta_t_pending_objects
   CHANGING
      p_spta_para_exp   TYPE spta_t_para_exp
      p_user_param      TYPE t_user_param.

  DATA:rfc_msg(80)    TYPE c,
       rfc_failed     TYPE c,
       l_subrc        LIKE sy-subrc,
       l_tabsize      LIKE sy-tabix,
       ln_pending_obj LIKE LINE OF p_pending_objs.
  DATA:itab_tmp LIKE TABLE OF zidpmmaging WITH HEADER LINE.

  REFRESH:itab_tmp.
  RECEIVE RESULTS FROM FUNCTION 'ZRFC_MM_IDP_0294'
       TABLES
            it_data = itab_tmp
     EXCEPTIONS
        system_failure        = 1 MESSAGE rfc_msg
        communication_failure = 2 MESSAGE rfc_msg
        OTHERS                = 3.

  l_subrc = sy-subrc.

  CASE l_subrc.
    WHEN 0.
      CLEAR p_spta_para_exp-rfc_subrc.
      APPEND LINES OF itab_tmp TO gt_display.
    WHEN 1.
      p_spta_para_exp-rfc_subrc = spta_rfc_subrc-system_failure.

    WHEN 2.
      p_spta_para_exp-rfc_subrc = spta_rfc_subrc-communication_failure.

    WHEN 3.

      DESCRIBE TABLE p_pending_objs LINES l_tabsize.
      IF l_tabsize > 1.
        p_spta_para_exp-rfc_subrc = spta_rfc_subrc-application_error.
      ELSE.
        p_spta_para_exp-rfc_subrc = spta_rfc_subrc-no_error.
        READ TABLE p_pending_objs INDEX 1 INTO ln_pending_obj.
      ENDIF.

    WHEN OTHERS.
      RAISE unexpected_subrc.

  ENDCASE.

ENDFORM.

*---------------------------------------------------------------------*
*       FORM send_ftp                                                 *
*---------------------------------------------------------------------*
*       ........                                                      *
*---------------------------------------------------------------------*
*  -->  P_LOCALFILE                                                   *
*---------------------------------------------------------------------*
FORM send_ftp USING p_localfile.
  DATA:dsn          LIKE ibipparms-path,
       filename(20) TYPE c VALUE 'C*',
       user(64)     TYPE c,
       pwd(64)      TYPE c,
       host(64)     TYPE c VALUE '10.156.4.96',
       key          TYPE i VALUE 26101957,
       dest         LIKE rfcdes-rfcdest VALUE 'SAPFTPA',
       dstlen       TYPE i,
       hdl          TYPE i.
  DATA:ftp_handle       TYPE i,
       ftp_command(265) TYPE c,
       ftp_result(100)  TYPE c OCCURS 0 WITH HEADER LINE.
  DATA:ftpfile(100) TYPE c,
       sdir(100),
       command(100),
       fn           TYPE rlgrap-filename, ldir(100).
  DATA:l_pwd        LIKE tczgl-ftp_passwd,
       l_pwd2       LIKE tczgl-ftp_passwd,
       l_collect(1) TYPE c.
  DATA:BEGIN OF lt_result OCCURS 0,
         line(255),
       END OF lt_result.

  DATA: slen TYPE i.

  host  = p_ftp.
  user  = p_user.

  CLEAR: l_pwd,l_pwd2.
  l_pwd = p_pwd.
*  CALL FUNCTION 'CV_STRING_SCRAMBLE'
*    EXPORTING
*      source = l_pwd
*      scr    = 'X'
*    IMPORTING
*      dest   = l_pwd2.
  slen = strlen( l_pwd ).
  CALL FUNCTION 'HTTP_SCRAMBLE'
    EXPORTING
      source      = l_pwd
      sourcelen   = slen
      key         = key
    IMPORTING
      destination = l_pwd2.
  pwd = l_pwd2.

  CLEAR:l_collect.
  DO 3 TIMES.
    CALL FUNCTION 'FTP_CONNECT'
      EXPORTING
        user            = user
        password        = pwd
        host            = host
        rfc_destination = 'SAPFTPA'  "R/3 application server
      IMPORTING
        handle          = ftp_handle
      EXCEPTIONS
        not_connected   = 1
        OTHERS          = 2.
    IF sy-subrc = 0.
      l_collect = 'Y'.
      EXIT.
    ELSE.
      WRITE:/ 'Connect FTP error'.
      CONTINUE.
    ENDIF.
  ENDDO.
  IF l_collect <> 'Y'.
    WRITE:/ 'Connect FTP error'.
    EXIT.
  ELSE.
    WRITE:/ 'Connect FTP success'.
  ENDIF.

  CONCATENATE 'cd' p_path INTO command SEPARATED BY ' '.
  CONDENSE command.
  CALL FUNCTION 'FTP_COMMAND'
    EXPORTING
      handle        = ftp_handle
      command       = command
    TABLES
      data          = lt_result
    EXCEPTIONS
      tcpip_error   = 1
      command_error = 2
      data_error    = 3
      OTHERS        = 4.
  LOOP AT lt_result.
    WRITE:/ lt_result-line.
    IF lt_result-line CS 'error'.
      EXIT.
    ENDIF.
  ENDLOOP.

  CONCATENATE 'put' p_localfile INTO command
              SEPARATED BY ' '.
  CONDENSE command.
  CALL FUNCTION 'FTP_COMMAND'
    EXPORTING
      handle        = ftp_handle
      command       = command
    TABLES
      data          = lt_result
    EXCEPTIONS
      tcpip_error   = 1
      command_error = 2
      data_error    = 3
      OTHERS        = 4.
  LOOP AT lt_result.
    WRITE:/ lt_result-line.
    IF lt_result-line CS 'error'.
      EXIT.
    ENDIF.
  ENDLOOP.

  CALL FUNCTION 'FTP_DISCONNECT'
    EXPORTING
      handle = ftp_handle.

  DELETE DATASET p_file.
ENDFORM.

*---------------------------------------------------------------------*
*       FORM openxls                                                  *
*---------------------------------------------------------------------*
*       ........                                                      *
*---------------------------------------------------------------------*
FORM openxls.
  CREATE OBJECT h_excel 'EXCEL.APPLICATION'.
  CALL METHOD OF h_excel 'Workbooks' = h_workbooks.
  CALL METHOD OF h_workbooks 'ADD' = h_workbook.
  SET PROPERTY OF h_excel 'Visible' = 0.
  GET PROPERTY OF h_excel 'ActiveSheet' = h_worksheet.
ENDFORM.

*---------------------------------------------------------------------*
*       FORM closexls                                                 *
*---------------------------------------------------------------------*
*       ........                                                      *
*---------------------------------------------------------------------*
FORM closexls.

  CALL METHOD OF h_workbook 'SAVEAS'
    EXPORTING
      #1 = p_filena.
  IF sy-subrc = 0.
    WRITE:/ 'Success'.
  ELSE.
    WRITE:/ 'Fail'.
  ENDIF.

  CALL METHOD OF h_excel 'QUIT'.
  FREE OBJECT h_worksheet.
  FREE OBJECT h_workbook.
  FREE OBJECT h_excel.
ENDFORM.

*---------------------------------------------------------------------*
*       FORM get_inventory2                                           *
*---------------------------------------------------------------------*
*       ........                                                      *
*---------------------------------------------------------------------*
FORM get_inventory2.
  DATA:BEGIN OF lt_inv OCCURS 0,
         matnr      LIKE mard-matnr,
         werks      LIKE mard-werks,
         bukrs      LIKE t001k-bukrs,
         lgort      LIKE mard-lgort,
         charg      LIKE mspr-charg,
         sobkz      LIKE mspr-sobkz,
         pspnr      LIKE mspr-pspnr,
         lfgja      LIKE mard-lfgja,
         lfmon      LIKE mard-lfmon,
         labst      LIKE mard-labst,
         insme      LIKE mard-insme,
         indicator3 LIKE zidpmm016-indicator3,
         depart     LIKE zidpmm016-depart,
         category1  LIKE zidpmm016-category1,
         category2  LIKE zidpmm016-category2,
         azf01      LIKE zidpmm017-azf03,
         azf02      LIKE zidpmm017-azf03,
         matkl      LIKE mara-matkl,
         wgbez      LIKE t023t-wgbez,
       END OF lt_inv.
  DATA:lt_t001k LIKE t001k OCCURS 0 WITH HEADER LINE.
  DATA:l_baseyear  LIKE mard-lfgja,
       l_basemonth LIKE mard-lfmon.
  DATA:l_month1     LIKE mard-lfmon,
       l_month2     LIKE mard-lfmon,
       l_month3     LIKE mard-lfmon,
       l_date       LIKE sy-datum,
       l_lastday(1) TYPE c.
  DATA:BEGIN OF lt_matnr OCCURS 0,
         werks LIKE marc-werks,
         matnr LIKE marc-matnr,
         charg LIKE mspr-charg,
         waers LIKE ckms_prs_ml-waers,
         stprs LIKE ckms_prs_ml-stprs,
         peinh LIKE ckms_prs_ml-peinh,
       END OF lt_matnr.
  DATA:lt_cki_prs LIKE ckms_prs_ml OCCURS 0 WITH HEADER LINE,
       lw_cki_pr1 TYPE cki_pr1_ml.
  DATA:l_bdatj LIKE ckmlpp-bdatj,
       l_poper LIKE ckmlpp-poper.
  DATA:l_bwtar      LIKE mbew-bwtar,
       l_indicator1 LIKE zidpmm016-indicator1,
       l_cvalue     LIKE zidpcontrolvalue-cvalue.


* 如果基準日期是上月最後一天,直接取期末庫存
* 如果基準日期是上月最後一天以前,取上一月期末庫存+當前庫存(置0)
* 如果基準日期是上月最後一天以後,取期末庫存+當前庫存(置0)
* 取基準日期的年月
  l_month3 = p_date+4(2).
  l_date = p_date + 1.
  l_month1 = l_date+4(2).
  l_date = sy-datum.
  l_month2 = l_date+4(2).
  IF l_month3 <> l_month1 AND l_month3 <> l_month2.
    l_lastday = 'Y'.   "上月最後一天
    l_baseyear = p_date+0(4).
    l_basemonth = p_date+4(2).
  ELSEIF l_month3 = l_month2.
    l_lastday = 'A'.  "當月，即上月最後一天以後

    CALL FUNCTION 'CCM_GO_BACK_MONTHS'
      EXPORTING
        currdate   = p_date
        backmonths = 1
      IMPORTING
        newdate    = l_date.
    l_baseyear = l_date+0(4).
    l_basemonth = l_date+4(2).
  ELSE.
    l_lastday = 'B'.  "上月最後一天以前

    CALL FUNCTION 'CCM_GO_BACK_MONTHS'
      EXPORTING
        currdate   = p_date
        backmonths = 1
      IMPORTING
        newdate    = l_date.
    l_baseyear = l_date+0(4).
    l_basemonth = l_date+4(2).
  ENDIF.
* 記錄最新時間
  g_date = sy-datum + 2.
  g_time = sy-uzeit.
* 抓取庫存
  IF NOT itab_016[] IS INITIAL.
    SELECT zidpfi_mv_sum~werks zidpfi_mv_sum~matnr zidpfi_mv_sum~lgort
           zidpfi_mv_sum~gjahr AS lfgja zidpfi_mv_sum~pspnr
* 增加主鍵，防止被自動去重 Add by wushaui 20220705
           zidpfi_mv_sum~gjahr AS lfmon
           menge AS labst
           mara~matkl
    INTO CORRESPONDING FIELDS OF TABLE lt_inv
    FROM zidpfi_mv_sum
    INNER JOIN mara ON mara~matnr = zidpfi_mv_sum~matnr
    FOR ALL ENTRIES IN itab_016
    WHERE zidpfi_mv_sum~werks = itab_016-werks
    AND zidpfi_mv_sum~lgort = itab_016-lgort
    AND zidpfi_mv_sum~matnr IN s_matnr
    AND mara~mtart IN s_mtart
    AND zidpfi_mv_sum~gjahr = l_baseyear
    AND zidpfi_mv_sum~monat = l_basemonth.

    IF l_lastday <> 'Y'.
      SELECT mard~matnr mard~werks mard~lgort mard~lfgja mard~lfmon
*             0 AS labst 0 AS insme
             mara~matkl
      APPENDING CORRESPONDING FIELDS OF TABLE lt_inv
      FROM mard
      INNER JOIN mara ON mara~matnr = mard~matnr
      FOR ALL ENTRIES IN itab_016
      WHERE mard~werks = itab_016-werks
      AND mard~lgort = itab_016-lgort
      AND mard~matnr IN s_matnr
      AND mara~mtart IN s_mtart.

      SELECT mspr~matnr mspr~werks mspr~lgort mspr~charg mspr~sobkz
             mspr~pspnr mspr~lfgja mspr~lfmon mara~matkl
*             0 AS labst
      APPENDING CORRESPONDING FIELDS OF TABLE lt_inv
      FROM mspr
      INNER JOIN mara ON mara~matnr = mspr~matnr
      FOR ALL ENTRIES IN itab_016
      WHERE mspr~werks = itab_016-werks
      AND mspr~lgort = itab_016-lgort
      AND mspr~matnr IN s_matnr
      AND mara~mtart IN s_mtart.

      IF p_nco = 'X'.
        SELECT mkol~matnr mkol~werks mkol~lgort mkol~charg mkol~sobkz
               mkol~lfgja mkol~lfmon mara~matkl
*               0 AS labst
        APPENDING CORRESPONDING FIELDS OF TABLE lt_inv
        FROM mkol
        INNER JOIN mara ON mara~matnr = mkol~matnr
        FOR ALL ENTRIES IN itab_016
        WHERE mkol~werks = itab_016-werks
        AND mkol~lgort = itab_016-lgort
        AND mkol~matnr IN s_matnr
        AND mara~mtart IN s_mtart.
      ENDIF.
    ENDIF.
  ENDIF.
* 取庫存中所有料號的標準成本
  IF p_nco = 'X'.
    LOOP AT lt_inv.
      lt_matnr-werks = lt_inv-werks.
      lt_matnr-matnr = lt_inv-matnr.
      lt_matnr-charg = lt_inv-charg.
      COLLECT lt_matnr.
    ENDLOOP.
    LOOP AT lt_matnr.
      CLEAR:l_bwtar.
      SELECT SINGLE cvalue INTO l_cvalue
      FROM zidpcontrolvalue
      WHERE werks = lt_matnr-werks
      AND ctype = 'VALUATION_CHECK'
      AND indicator1 = 'ACT'.
      IF sy-subrc = 0.
        IF lt_matnr-charg <> ''.
          l_bwtar = lt_matnr-charg.
        ELSE.
          SELECT SINGLE bwtar INTO l_bwtar
          FROM mbew
          WHERE matnr = lt_matnr-matnr
          AND bwkey = lt_matnr-werks
          AND bwtar LIKE 'COST%'.
        ENDIF.
      ELSE.
        l_bwtar = ''.
      ENDIF.

      CLEAR:lt_cki_prs,lt_cki_prs[],lw_cki_pr1.
      CLEAR:l_bdatj,l_poper.
      lw_cki_pr1-matnr = lt_matnr-matnr.
      lw_cki_pr1-bwkey = lt_matnr-werks.
      lw_cki_pr1-bwtar = l_bwtar.
      l_bdatj = l_baseyear.
      l_poper = l_basemonth.

      CALL FUNCTION 'Z_CKMS_MATERIAL_PRICES_GET'
        EXPORTING
          i_bdatj_1          = l_bdatj
          i_poper_1          = l_poper
        TABLES
          t_cki_prs          = lt_cki_prs
        CHANGING
          f_cki_pr1          = lw_cki_pr1
        EXCEPTIONS
          material_not_found = 1
          insufficient_input = 2
          OTHERS             = 3.
      IF NOT lt_cki_prs[] IS INITIAL.
        READ TABLE lt_cki_prs INDEX 1.
        lt_matnr-waers = lt_cki_prs-waers.
        lt_matnr-stprs = lt_cki_prs-stprs.
        lt_matnr-peinh = lt_cki_prs-peinh.
        MODIFY lt_matnr.
      ENDIF.
    ENDLOOP.
  ENDIF.
* 讀取法人信息到內表
  SELECT * INTO CORRESPONDING FIELDS OF TABLE lt_t001k
  FROM t001k
  WHERE bukrs IN s_bukrs.

  SELECT * INTO CORRESPONDING FIELDS OF TABLE itab_023
  FROM t023t.

  LOOP AT lt_inv.
    READ TABLE lt_t001k WITH KEY bwkey = lt_inv-werks.
    IF sy-subrc = 0.
      lt_inv-bukrs = lt_t001k-bukrs.
    ENDIF.

    CLEAR:l_indicator1.
    READ TABLE itab_016 WITH KEY werks = lt_inv-werks
                                 lgort = lt_inv-lgort.
    IF sy-subrc = 0.
      lt_inv-indicator3 = itab_016-indicator3.
      lt_inv-depart = itab_016-depart.
      lt_inv-category1 = itab_016-category1.
      lt_inv-category2 = itab_016-category2.
      l_indicator1 = itab_016-indicator1.
    ENDIF.

    READ TABLE  itab_017 WITH KEY werks = lt_inv-werks
                                  azf01 = itab_016-category1.
    IF sy-subrc = 0 .
      lt_inv-azf01 = itab_017-azf03.
    ENDIF.

    READ TABLE  itab_017 WITH KEY werks = lt_inv-werks
                                  azf01 = itab_016-category2.
    IF sy-subrc = 0 .
      lt_inv-azf02 = itab_017-azf03.
    ENDIF.

    READ TABLE  itab_023 WITH KEY matkl = lt_inv-matkl.
    IF sy-subrc = 0 .
      lt_inv-wgbez = itab_023-wgbez.
    ENDIF.

    gt_display-matnr = lt_inv-matnr.
    gt_display-bu = lt_inv-bukrs.
    gt_display-werks = lt_inv-werks.
    gt_display-lgort = lt_inv-lgort.
    gt_display-charg = lt_inv-charg.
    gt_display-sobkz = lt_inv-sobkz.
    gt_display-wbs = lt_inv-pspnr.
    gt_display-total = lt_inv-labst.
    gt_display-indicator3 = lt_inv-indicator3.
    gt_display-depart = lt_inv-depart.
    gt_display-azf01 = lt_inv-azf01.
    gt_display-azf02 = lt_inv-azf02.
    gt_display-matkl = lt_inv-matkl.
    gt_display-wgbez = lt_inv-wgbez.

*   如果庫存為0,檢查基準日之後是否有異動
    IF gt_display-total = 0.
      IF l_baseyear > lt_inv-lfgja OR
      ( l_baseyear = lt_inv-lfgja AND l_basemonth > lt_inv-lfmon ).
        CONTINUE.   "基準日期月份之後未異動的不考慮
      ENDIF.
    ENDIF.
*   庫存金額
    CLEAR:lt_matnr.
    READ TABLE lt_matnr WITH KEY werks = gt_display-werks
                                 matnr = gt_display-matnr.
    IF sy-subrc = 0.
      gt_display-waers = lt_matnr-waers.
      IF gt_display-total > 0 AND l_indicator1 = 'Y' AND p_nco = 'X'.
        gt_display-totval =
        gt_display-total * lt_matnr-stprs / lt_matnr-peinh.
      ENDIF.
    ENDIF.
*   level合併
    IF p_bu = 'X'.
      gt_display-werks = ''.
      gt_display-lgort = ''.
      gt_display-charg = ''.
      gt_display-sobkz = ''.
      gt_display-wbs = ''.
      gt_display-indicator3 = ''.
      gt_display-depart = ''.
      gt_display-azf01 = ''.
      gt_display-azf02 = ''.
      gt_display-matkl = ''.
      gt_display-wgbez = ''.
    ELSEIF p_pla = 'X'.
      gt_display-lgort = ''.
      gt_display-charg = ''.
      gt_display-sobkz = ''.
      gt_display-wbs = ''.
      gt_display-indicator3 = ''.
      gt_display-depart = ''.
      gt_display-azf01 = ''.
      gt_display-azf02 = ''.

    ELSEIF p_loc = 'X'.

    ENDIF.

    COLLECT gt_display.

*---------------------------------------------------------------------*
*       FORM Global_cache                                             *
*---------------------------------------------------------------------*
*       ........                                                      *
*---------------------------------------------------------------------*
FORM global_cache.
  DATA:exch_rate LIKE bapi1093_0.
  DATA: ll_count TYPE i.
  DATA: ll_date LIKE sy-datum.
  DATA: ll_jobn LIKE tbtco-jobname.
  DATA:zglobal LIKE zidpmmaging_glo OCCURS 0 WITH HEADER LINE.

*排程檢查，只能固定排程名稱，才允許存數據
  IF sy-batch = 'X'.
    CLEAR ll_jobn.
    CONCATENATE '%' 'MM_GLOBAL_003C' p_job '%' INTO ll_jobn.
    ll_date = sy-datum - 2.
    SELECT COUNT( DISTINCT jobcount ) INTO ll_count FROM tbtco
    WHERE jobname LIKE ll_jobn
    AND status = 'R'
    AND sdlstrtdt > ll_date.
    IF ll_count <> 1.
      MESSAGE i004(zmm) WITH 'exist job of active'.
      STOP.
    ENDIF.
  ELSE.
    MESSAGE i004(zmm) WITH '當前程式不支持手動執行'.
    STOP.
  ENDIF.

*全部滿足，則保存
  IF NOT gt_display[] IS INITIAL.
    LOOP AT gt_display.
      MOVE-CORRESPONDING gt_display TO zglobal.
*      zglobal-transdate = sy-datum.
      zglobal-transdate = p_date.
      zglobal-waers1 = 'NTD'.
      zglobal-transtime = sy-uzeit.
      zglobal-flag = 'N'.
*轉換為NDT
      IF gt_display-value13 <> '0.00'.
        CALL FUNCTION 'BAPI_EXCHANGERATE_GETDETAIL'
          EXPORTING
            rate_type  = 'M'
            from_curr  = gt_display-waers
            to_currncy = 'NTD'
            date       = p_date
          IMPORTING
            exch_rate  = exch_rate.
        zglobal-value13_a = gt_display-value13 * exch_rate-exch_rate.
      ELSE.
        zglobal-flag = 'V'.
      ENDIF.
      APPEND zglobal.
      CLEAR:gt_display,zglobal.
    ENDLOOP.
*更新表裡
    MODIFY zidpmmaging_glo FROM TABLE zglobal.
    IF sy-subrc = 0.
      WRITE:/ 'Success'.
    ELSE.
      WRITE:/ 'Fail'.
    ENDIF.
  ENDIF.
ENDFORM.
