CLASS lcl_handle_events DEFINITION.
  PUBLIC SECTION.
    METHODS:
      on_link_click FOR EVENT link_click OF cl_salv_events_table
        IMPORTING row column.
ENDCLASS.

CLASS lcl_handle_events IMPLEMENTATION.
  METHOD on_link_click.

    READ TABLE itab INDEX row ASSIGNING FIELD-SYMBOL(<row>).
    IF sy-subrc = 0.
      ASSIGN COMPONENT column OF STRUCTURE <row> TO FIELD-SYMBOL(<value>).
    ENDIF.

    IF column = 'ZCONFIG' AND <value> IS NOT INITIAL.
      cl_demo_output=>display( <value> ).
    ENDIF.

  ENDMETHOD.                    "on_single_click
ENDCLASS.

FORM getdata.
  DATA : l_mod      LIKE zidpmmcontrol-value1,
         l_model    LIKE zidppp178-product,
         l_qpa      LIKE zidppp178-menge,
         l_qpa2(10) TYPE c,

  DATA:l_code      LIKE sy-subrc,
       l_valuation LIKE mbew-bwtar,
       l_mes       LIKE syst-msgv1.
  DATA:ls_dis TYPE ty_dis.

  IF p_la = 'X'.                                            "stock > 0
    SELECT mard~lgort mard~matnr  mard~labst
    INTO CORRESPONDING FIELDS OF TABLE itab_matnr
    FROM mard
    WHERE mard~matnr IN p_matnr AND mard~werks = p_plant
    AND mard~lgort IN p_sloc AND mard~labst > 0.
  ELSE.
    SELECT mard~lgort mard~matnr  mard~labst
    INTO CORRESPONDING FIELDS OF TABLE itab_matnr
    FROM mard
    WHERE mard~matnr IN p_matnr AND mard~werks = p_plant
    AND mard~lgort IN p_sloc.
  ENDIF .

  IF p_spe = 'X'. "Special stock
    IF p_la = 'X'.                                          "Stock>0
      SELECT mkol~lgort mkol~matnr mkol~slabs AS labst
      APPENDING CORRESPONDING FIELDS OF TABLE itab_matnr
      FROM mkol
      WHERE mkol~matnr IN p_matnr AND mkol~werks = p_plant
      AND mkol~lgort IN p_sloc AND mkol~slabs > 0.

      SELECT mspr~lgort mspr~matnr mspr~prlab AS labst
      mspr~pspnr
      APPENDING CORRESPONDING FIELDS OF TABLE itab_matnr
      FROM mspr
      WHERE mspr~matnr IN p_matnr AND mspr~werks = p_plant
      AND mspr~lgort IN p_sloc AND mspr~pspnr IN p_wbs
      AND mspr~prlab > 0.
    ELSE.
      SELECT mkol~lgort mkol~matnr mkol~slabs AS labst
      APPENDING CORRESPONDING FIELDS OF TABLE itab_matnr
      FROM mkol
      WHERE mkol~matnr IN p_matnr AND mkol~werks = p_plant
      AND mkol~lgort IN p_sloc.

      SELECT mspr~lgort mspr~matnr mspr~prlab AS labst
      mspr~pspnr
      APPENDING CORRESPONDING FIELDS OF TABLE itab_matnr
      FROM mspr
      WHERE mspr~matnr IN p_matnr AND mspr~werks = p_plant
      AND mspr~lgort IN p_sloc AND mspr~pspnr IN p_wbs.
    ENDIF .
  ENDIF .
