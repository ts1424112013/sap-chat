FORM classify_status_table.
  SORT itab BY bstnk poitem bstdk vbeln posnr.
  IF so_all EQ 'X'.
    itab_all = itab[].
    itab_alv[]  = itab_all[].

  ELSEIF so_open EQ 'X'.
*    LOOP AT itab INTO line WHERE qty2 <> 0 AND dn GE ''.
    LOOP AT itab INTO line.
      IF line-qty2 <> 0 AND line-dn EQ ''.
        APPEND line TO itab_so_open.
      ENDIF.
    ENDLOOP.
    itab_alv[]  = itab_so_open[].

  ELSEIF dn_open EQ 'X'.
    LOOP AT itab INTO line  WHERE dn NE '' AND ( pgidate EQ '' OR
 pgidate EQ '00000000' ).
*        ( pgidate EQ '00000000' OR pgidate EQ '' ) .
      APPEND line TO itab_dn_open.
    ENDLOOP.
    itab_alv[]  = itab_dn_open[].

  ELSEIF so_close EQ 'X'.
*    LOOP AT itab INTO line.
*      IF itab-qty1 = itab-soqty.
*        APPEND line TO itab_so_close.
*      ENDIF.
*    ENDLOOP.
    LOOP AT itab INTO line.
      IF line-billno <> ''.
        APPEND line TO itab_so_close.
      ENDIF.
    ENDLOOP.
    itab_alv[]  = itab_so_close[].
  ELSEIF pgi_open EQ 'X'.
    LOOP AT itab INTO line WHERE pgidate <> '' AND billno = ''
                            AND pgidate <> '00000000'.
      APPEND line TO itab_pgi_open.
    ENDLOOP.
    itab_alv[]  = itab_pgi_open[].
  ENDIF.
ENDFORM.                    "CLASSIFY_STATUS_TABLE
