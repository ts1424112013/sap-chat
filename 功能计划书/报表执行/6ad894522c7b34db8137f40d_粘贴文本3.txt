*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM sel_data.
  SELECT *
  INTO corresponding FIELDS OF TABLE itab
  FROM zidpmm016 AS a
  inner JOIN t001l AS b ON a~werks = b~werks
  AND a~lgort = b~lgort
  WHERE a~werks = p_werks
  AND a~lgort IN p_lgort.
  IF itab[] IS INITIAL.
    MESSAGE i000(zmm) WITH 'No Data Display'.
    STOP.
  ENDIF.
  LOOP AT itab.
    SELECT SINGLE azf03 INTO itab-category1des FROM zidpmm017
    WHERE werks = itab-werks AND azf01 = itab-category1.
    SELECT SINGLE azf03 INTO itab-category2des FROM zidpmm017
    WHERE werks = itab-werks AND azf01 = itab-category2.
    SELECT SINGLE azf03 INTO itab-category4des FROM zidpmm017
    WHERE werks = itab-werks AND azf01 = itab-category4.
    SELECT SINGLE azf03 INTO itab-category5des FROM zidpmm017
    WHERE werks = itab-werks AND azf01 = itab-category5.
    SELECT SINGLE cvalue1 INTO itab-build_code FROM zidpcontrolvalue
    WHERE werks = itab-werks
    AND ctype = 'BUILD_CODE'
    AND cvalue = itab-category6+0(3)
    AND indicator1 = 'ACT'.
    MODIFY itab.
    CLEAR itab.
  ENDLOOP.
ENDFORM.                    " sel_data

FORM check_authorization.
  RANGES: l_werks FOR marc-werks.
  l_werks-SIGN = 'I'.
  l_werks-option = 'EQ'.
  l_werks-low = p_werks.
  APPEND l_werks.
  CLEAR l_werks.
  CALL FUNCTION 'ZRFC_MM_IDP_0104'
  TABLES
    INPUT  = l_werks[]
    OUTPUT = l_werks[].
  IF l_werks[] IS INITIAL.
    IF sy-langu = 'E'.
      MESSAGE i004(zmm) WITH 'You have no authority to use this plant'
      .
    ELSE.
      MESSAGE i004(zmm) WITH '您沒有查詢這個工廠的權限'.
    ENDIF.
    STOP.
  ENDIF.
ENDFORM.                    " check_authorization
