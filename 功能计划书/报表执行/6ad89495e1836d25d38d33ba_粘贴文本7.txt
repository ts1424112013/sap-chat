    SORT lt_mseg2 BY smbln.
  ENDIF.

  lt_lgort[] = VALUE ty_lgort( FOR ls_wh IN zidpmmwhhtab[] ( werks = ls_wh-werks lgort = ls_wh-lgort )
                                                        ( werks = ls_wh-umwrk lgort = ls_wh-umlgo ) ).
  SORT lt_lgort BY werks lgort.
  DELETE ADJACENT DUPLICATES FROM lt_lgort COMPARING ALL FIELDS.

  SELECT a~werks, a~lgort, a~category6
  FROM zidpmm016 AS a
  FOR ALL ENTRIES IN @lt_lgort
  WHERE a~werks = @lt_lgort-werks
  AND a~lgort = @lt_lgort-lgort
  INTO TABLE @DATA(lt_016).

  SELECT domname, domvalue_l,ddtext
  INTO TABLE @DATA(lt_dd07t)
  FROM dd07t
  WHERE domname IN ('ZDOCTYPE','ZCUSMARK')
  AND ddlanguage = @sy-langu.
  SORT lt_dd07t BY domname domvalue_l.

  SORT zidpmmwhhtab BY docnumber docitem.
  LOOP AT zidpmmwhhtab.
    READ TABLE lt_dd07t INTO DATA(ls_dd07t) WITH KEY domname = 'ZDOCTYPE'
                                                     domvalue_l = zidpmmwhhtab-doctype BINARY SEARCH.
    IF sy-subrc = 0.
      zidpmmwhhtab-doctypetext = ls_dd07t-ddtext.
    ENDIF.

    IF zidpmmwhhtab-doctype = 'D' AND ( zidpmmwhhtab-werks CS 'IN' OR zidpmmwhhtab-werks CS 'KA' ).
      READ TABLE lt_016 INTO DATA(ls_016) WITH KEY werks = zidpmmwhhtab-werks
                                                   lgort = zidpmmwhhtab-lgort BINARY SEARCH.
      IF sy-subrc = 0.
        zidpmmwhhtab-build = ls_016-category6+0(3).
      ENDIF.
      READ TABLE lt_016 INTO ls_016 WITH KEY werks = zidpmmwhhtab-werks
                                             lgort = zidpmmwhhtab-umlgo BINARY SEARCH.
      IF sy-subrc = 0.
        zidpmmwhhtab-build1 = ls_016-category6+0(3).
      ENDIF.
    ENDIF.

    IF zidpmmwhhtab-doctype CA 'GH'.
      READ TABLE lt_dd07t INTO ls_dd07t WITH KEY domname = 'ZCUSMARK'
                                                 domvalue_l = zidpmmwhhtab-cusremark BINARY SEARCH.
      IF sy-subrc = 0.
        zidpmmwhhtab-cusremarktxt = ls_dd07t-ddtext.
      ENDIF.
    ENDIF .

    IF NOT zidpmmwhhtab-mblnr IS INITIAL .
      READ TABLE lt_mseg2 INTO DATA(ls_mseg2) WITH KEY smbln = zidpmmwhhtab-mblnr BINARY SEARCH.
      IF sy-subrc = 0.
        zidpmmwhhtab-roll = 'X' .
        zidpmmwhhtab-smbln = ls_mseg2-mblnr .
      ENDIF .

      READ TABLE lt_mseg INTO DATA(ls_mseg) WITH KEY mblnr = zidpmmwhhtab-mblnr
                                                     matnr = zidpmmwhhtab-matnr
                                                     erfmg = zidpmmwhhtab-erfmg
                                                     werks = zidpmmwhhtab-werks
                                                     lgort = zidpmmwhhtab-lgort
                                                     umlgo = zidpmmwhhtab-umlgo BINARY SEARCH.
      IF sy-subrc = 0.
        zidpmmwhhtab-zeile = ls_mseg-zeile.
      ENDIF.

      READ TABLE lt_mseg INTO DATA(ls_mseg1) WITH KEY mblnr = zidpmmwhhtab-mblnr BINARY SEARCH.
      IF sy-subrc = 0.
        zidpmmwhhtab-cpudt = ls_mseg1-cpudt.
      ENDIF.
    ENDIF .

    READ TABLE lt_mmctl2 TRANSPORTING NO FIELDS WITH KEY werks = zidpmmwhhtab-werks BINARY SEARCH.
    IF sy-subrc = 0.
      CLEAR:l_bwkey,l_bwtar,l_kalnr,l_pspnr.
      l_poper_1 = sy-datum+4(2).
      l_bdatj_1 = sy-datum+0(4).

      SELECT SINGLE bukrs FROM t001k INTO l_bukrs
      WHERE bwkey = zidpmmwhhtab-werks .

      SELECT SINGLE waers FROM t001 INTO l_waers
      WHERE bukrs = l_bukrs .

      CALL FUNCTION 'CURRENCY_CONVERTING_FACTOR'
        EXPORTING
          currency = l_waers
        IMPORTING
          factor   = l_factor.

      SELECT SINGLE bwkey bwtar FROM mbew INTO (l_bwkey,l_bwtar)
      WHERE matnr = zidpmmwhhtab-matnr
      AND bwkey = zidpmmwhhtab-werks.
      SELECT SINGLE kalnr pspnr FROM ckmlhd INTO (l_kalnr,l_pspnr)
      WHERE matnr = zidpmmwhhtab-matnr
      AND bwkey = l_bwkey
      AND bwtar = l_bwtar.

      CLEAR:gt_kalnr,gt_kalnr[],gt_ckmlpp,gt_ckmlpp[], gt_ckmlcr,gt_ckmlcr[].
      gt_kalnr-kalnr = l_kalnr.
      gt_kalnr-bwkey = l_bwkey.
      gt_kalnr-bwtar = l_bwtar.
      gt_kalnr-pspnr = l_pspnr.
      APPEND gt_kalnr.CLEAR gt_kalnr.

      CALL FUNCTION 'CKMS_PERIOD_READ_WITH_ITAB'
        EXPORTING
          i_use_buffer            = 'X'
          i_bdatj_1               = l_bdatj_1
          i_poper_1               = l_poper_1
        TABLES
          t_kalnr                 = gt_kalnr
          t_ckmlpp                = gt_ckmlpp
          t_ckmlcr                = gt_ckmlcr
        EXCEPTIONS
          no_data_found           = 1
          input_data_inconsistent = 2
          buffer_inconsistent     = 3
          OTHERS                  = 4.


      zidpmmwhhtab-tslvt = gt_ckmlcr-stprs * l_factor.
      zidpmmwhhtab-tltslvt = zidpmmwhhtab-tslvt * zidpmmwhhtab-erfmg.
    ENDIF.

    MODIFY zidpmmwhhtab .
  ENDLOOP.
ENDFORM.
