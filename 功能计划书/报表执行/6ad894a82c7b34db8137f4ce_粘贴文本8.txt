FORM get_mat  USING p_ucomm    LIKE sy-ucomm
                    p_selfield TYPE slis_selfield.

  DATA: lc_value(60) TYPE c.
  CASE p_ucomm.
    WHEN '&IC1'.
      READ TABLE itab1 INDEX p_selfield-tabindex.
      MOVE-CORRESPONDING itab1 TO  itab.
*      loop at itab .
      SELECT SINGLE value2 INTO itab-tlk51 FROM zidpmmcontrol
      WHERE werks = itab-werks
        AND project = 'RELATION'
        AND value1 = itab-tlk03 .
*      modify itab .
*      clear itab .
*      endloop.
*      CLEAR:ITAB-TLK32, ITAB-TLK28, ITAB-TLK26, ITAB-TLK14.
      CALL SCREEN 2000 .

    WHEN 'BACK' OR 'EXIT' OR 'CANCEL'.
      LEAVE TO SCREEN 0.
*  Begin Add by Denny for DV2K968292
    WHEN 'REFR'.
      CLEAR: itab1, itab.
      FREE: itab1, itab.
      PERFORM get_display_data.
      PERFORM write_alv_data.
      PERFORM call_alv_for_list.
  ENDCASE.

  IF g_flag = 'Y'.
    CLEAR: itab1, itab.
    FREE: itab1, itab.
    PERFORM get_display_data.
    PERFORM write_alv_data.
    PERFORM call_alv_for_list.
    g_flag = 'N'.
  ENDIF.
* end Add by Denny for DV2K968292

ENDFORM.
