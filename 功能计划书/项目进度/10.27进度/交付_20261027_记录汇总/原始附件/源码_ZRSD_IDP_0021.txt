    WHEN '&CREATE'.
      PERFORM cretran.

    WHEN 'BACK'.
      LEAVE SCREEN.
    WHEN 'EXIT'.
      LEAVE SCREEN.

      PERFORM sel_all.
    WHEN '&DESEL_ALL'.

  ENDCASE.

FORM cretran.
  DATA:gm_code   LIKE bapi2017_gm_code OCCURS 0 WITH HEADER LINE,
       gm_header LIKE bapi2017_gm_head_01 OCCURS 0 WITH HEADER LINE,
       gm_item   LIKE bapi2017_gm_item_create OCCURS 0 WITH HEADER LINE,
       it_return LIKE bapiret2 OCCURS 0 WITH HEADER LINE.
  DATA:l_matedoc LIKE mkpf-mblnr,subrc LIKE sy-subrc.

  DATA: lt_hubin LIKE zidpsd066 OCCURS 0 WITH HEADER LINE.

    ikk1 = ikk.
    APPEND ikk1.
  ENDLOOP.

  DELETE ADJACENT DUPLICATES FROM ikk1.
  CLEAR :ikk1.

  CLEAR: gm_code,gm_header,gm_item,it_return.
  REFRESH: gm_code,gm_header,gm_item,it_return.

  gm_code-gm_code = '04'.

  gm_header-doc_date = sy-datum.
  APPEND gm_header.

  LOOP AT ikk1.
*    gm_item-material = ikk1-matnr.
    gm_item-material_long = ikk1-matnr."Modify by Liv 20220503 使用40位物料字段
    gm_item-entry_qnt = ikk1-fqty.

    gm_item-plant =  ikk1-werks.

    CLEAR: gm_item-material_long,gm_item-entry_qnt."Modify by Liv 20220503 使用40位物料字段
  ENDLOOP.
  CLEAR gm_item .
  IF NOT gm_item[] IS INITIAL.
    CALL FUNCTION 'BAPI_GOODSMVT_CREATE'
      EXPORTING
        goodsmvt_header  = gm_header
        goodsmvt_code    = gm_code
      IMPORTING
        materialdocument = l_matedoc
      TABLES
        goodsmvt_item    = gm_item
        return           = it_return.

          wait = 'Y'.

    ikk1-mblnr = l_matedoc.
    ikk1-message = it_return-message.
    MODIFY ikk1.
  ENDLOOP.
  CLEAR gm_item.
  FREE gm_item.
