FORM get_data.
  TYPES: BEGIN OF ty_loadno,
           loadno TYPE zidpsd046-loadno,
         END OF ty_loadno.

  DATA: lt_kna1   TYPE STANDARD TABLE OF kna1,
        lt_loadno TYPE STANDARD TABLE OF ty_loadno,
        lt_scflag TYPE STANDARD TABLE OF ty_loadno,
        ls_loadno TYPE ty_loadno,
        lv_loadno TYPE zidpsd046-loadno.

  REFRESH t_dn.

  WITH
          +vbfa AS (
            SELECT vbelv,
                   MAX( vbeln ) AS vbeln,
                   MAX( erzet ) AS erzet,
                   MAX( posnn ) AS posnn
            FROM vbfa
            WHERE vbtyp_n = 'R'
            GROUP BY vbelv
          ),
          +vbrp_max AS (
            SELECT vbrp~vgbel, vbrp~posnr,  vbrp~vbeln
            FROM vbrp
            INNER JOIN vbrk ON vbrk~vbeln = vbrp~vbeln
            WHERE vbrk~fksto = ''
              AND vbrk~fkart NOT IN ( 'S1', 'S2', 'S3', 'ZS1', 'ZS2' )
              AND vbrp~draft = ''
              AND vbrk~draft = ''
          ),
          +zbsi_max AS (
            SELECT bsi12, MAX( bsi01 ) AS bsi01, MAX( decno ) AS decno
            FROM zbsi
            WHERE decno <> ''
            GROUP BY bsi12
          ),
          +zbsh_join AS (
            SELECT bsh01, MAX( bsh28 ) AS bsh28
            FROM zbsh
            GROUP BY bsh01
          ),
          +bkpf_max AS (
            SELECT xblnr, MAX( belnr ) AS belnr, MAX( gjahr ) AS gjahr, MAX( bukrs ) AS bukrs
            FROM bkpf
            WHERE xblnr <> ''
            GROUP BY xblnr
          ),
          +z017_max AS (
            " 將 zidpsd017 去重，確保每組 werks 和 ztll01 只有一筆包裝模式
            SELECT werks, ztll01, MAX( ztll902 ) AS ztll902
            FROM zidpsd017
            GROUP BY werks, ztll01
          ),
          +zctbs_max AS (
            " 將 zctbs_zzstatus 依 docno + item 去重，每組只保留一筆
            SELECT docno, item,
                   MAX( flag1 )   AS flag1,
                   MAX( rve2 )    AS rve2,
                   MAX( bsi03 )   AS bsi03,
                   MAX( bsi011 )  AS bsi011,
                   MAX( bsi02 )   AS bsi02
            FROM zctbs_zzstatus
            GROUP BY docno, item
          )

        SELECT
          likp~vbeln,
          likp~erdat,
          likp~wadat_ist,
          likp~kunnr,
          likp~bolnr,
          likp~lifex,
          likp~ernam,
          likp~aenam,
          likp~wbstk,
          lips~posnr,
          lips~matnr,
          lips~kdmat,
          lips~lfimg,
          lips~meins,
          lips~werks,
          lips~lgort,
          lips~vgbel,
          lips~vgpos,
          vbkd~kdgrp,
          vbkd~bstkd,
          vbkd~vsart,
          vbkd~ihrez,
          vbak~bname,
          vbak~bstzd,
          makt~maktx,
          zmt~tlk03 AS model,
          zmt~tlk45,
          zmt~tlk05,
          z017~ztll902 AS packmode,
          CASE WHEN z004~loadlistno IS NOT NULL THEN 'Y' ELSE 'N' END AS edi856flag,
          +vbfa~vbeln AS pgi,
          +vbfa~erzet AS erzet,
          +vbfa~posnn AS posnn,
          bkpf~usnam AS sapid,
          bkpf~xref1_hd AS employid,
          vbap~ps_psp_pnr AS wbs,
          zctbs~flag1,
          zctbs~rve2,
          zctbs~bsi03,
          zctbs~bsi011,
          zctbs~bsi02,
          +zbsi_max~decno AS decno,
          +zbsh_join~bsh28 AS bsh28,
          z016~category6 AS floor,
          z082~pmisc_flag,
          z082~misc_flag,
          z082~docnumber,
          tvlkt~vtext,
          +vbrp_max~vbeln AS billing
          FROM likp
          INNER JOIN lips ON lips~vbeln = likp~vbeln
          LEFT JOIN vbkd ON vbkd~vbeln = lips~vgbel AND vbkd~posnr = '000000'
          LEFT JOIN vbak ON vbak~vbeln = lips~vgbel
          LEFT JOIN makt ON makt~matnr = lips~matnr AND makt~spras = @sy-langu
          LEFT JOIN zidpmmtlk AS zmt ON zmt~tlk01 = lips~matnr AND zmt~werks = lips~werks
          LEFT JOIN +z017_max AS z017 ON z017~werks = lips~werks AND z017~ztll01 = vbkd~bstkd
          LEFT JOIN zidpsd004 AS z004 ON z004~werks = lips~werks AND z004~loadlistno = vbkd~bstkd AND z004~ftudn = likp~vbeln
          LEFT JOIN +vbfa ON +vbfa~vbelv = likp~vbeln
          LEFT JOIN t001k ON t001k~bwkey = lips~werks
          LEFT JOIN +bkpf_max ON +bkpf_max~xblnr = likp~vbeln
          LEFT JOIN bkpf ON bkpf~belnr = +bkpf_max~belnr
                        AND bkpf~gjahr = +bkpf_max~gjahr
                        AND bkpf~bukrs = +bkpf_max~bukrs
          LEFT JOIN vbap ON vbap~vbeln = lips~vgbel AND vbap~posnr = lips~vgpos
          LEFT JOIN +zctbs_max AS zctbs ON zctbs~docno = likp~vbeln AND zctbs~item = ltrim( lips~posnr, '0' )
          LEFT JOIN +zbsi_max ON +zbsi_max~bsi12 = likp~vbeln
          LEFT JOIN +zbsh_join ON +zbsh_join~bsh01 = +zbsi_max~bsi01
          LEFT JOIN zidpmm016 AS z016 ON z016~werks = lips~werks AND z016~lgort = lips~lgort
          LEFT JOIN zidpsd082 AS z082 ON z082~werks = lips~werks AND z082~vbeln = likp~vbeln
          LEFT JOIN +vbrp_max ON +vbrp_max~vgbel = likp~vbeln AND +vbrp_max~posnr = lips~posnr
          LEFT JOIN tvlkt ON tvlkt~spras = @sy-langu AND tvlkt~lfart = likp~lfart
          WHERE likp~vbeln IN @s_vbeln
            AND likp~lfart IN @s_lfart
            AND likp~erdat IN @s_erdat
            AND likp~wadat_ist IN @s_wadat
            AND likp~kunnr IN @s_kunnr
            AND lips~werks IN @p_plant
            AND lips~lgort IN @s_lgort
            AND likp~xabln IN @s_xabln
            AND ( zmt~tlk03 IN @s_model )
            AND ( vbkd~bstkd IN @s_bstkd )
            AND ( zmt~tlk05 IN @s_tlk05 )
            AND ( lips~matnr IN @s_matnr )
            AND ( vbap~ps_psp_pnr IN @s_wbs ) INTO CORRESPONDING FIELDS OF TABLE @t_dn.

  LOOP AT t_dn.
    SELECT SINGLE * FROM vbak WHERE vbeln = t_dn-vgbel.
*      SELECT SINGLE * FROM vbap WHERE vbeln = t_dn-vgbel
*      AND matnr = zloadi-matnr.
    SELECT SINGLE kbetr waers kpein kmein
    INTO (t_dn-kbetr,t_dn-konwa,t_dn-kpein,t_dn-kmein)
    FROM v_konv_cds
      "将KONV换成V_KONV_CDS  modify by ghm 20220428
    WHERE knumv  = vbak-knumv
    AND kposn  = t_dn-vgpos
    AND kschl = 'ZPR0'
      AND kinak NE 'M'"Add by lily 20230522.
      AND kinak NE 'Y'."Add by lily 202301220.
    MODIFY t_dn.
  ENDLOOP.


  IF sy-subrc <> 0.
    REFRESH t_dn.
    RETURN.
  ENDIF.

  " --- Apply radio button filters ---
  IF p1 = 'X'.
    DELETE t_dn WHERE wbstk <>  'C'.
  ELSEIF p2 = 'X'.
    DELETE t_dn WHERE wbstk = 'C'.
  ENDIF.

  IF q1 = 'X'.
    DELETE t_dn WHERE packmode <> 'S'.
  ELSEIF q2 = 'X'.
    DELETE t_dn WHERE packmode <> 'M'.
  ENDIF.

  " --- CONTYPE calculation ---
  LOOP AT t_dn ASSIGNING FIELD-SYMBOL(<fs_dn>).
    IF <fs_dn>-bstzd = 'DPKS' AND <fs_dn>-bstkd IS NOT INITIAL.
      <fs_dn>-contype = <fs_dn>-bstkd+4(3).
    ELSE.
      CLEAR <fs_dn>-contype.
    ENDIF.
  ENDLOOP.

  " --- Land1 batch fetch ---
  IF t_dn[] IS NOT INITIAL.
    SELECT kunnr, land1 FROM kna1
      FOR ALL ENTRIES IN @t_dn
      WHERE kunnr = @t_dn-kunnr
      INTO CORRESPONDING FIELDS OF TABLE @lt_kna1.
    SORT lt_kna1 BY kunnr.
    LOOP AT t_dn ASSIGNING <fs_dn>.
      READ TABLE lt_kna1 ASSIGNING FIELD-SYMBOL(<fs_kna>) WITH KEY kunnr = <fs_dn>-kunnr BINARY SEARCH.
      IF sy-subrc = 0.
        <fs_dn>-land1 = <fs_kna>-land1.
      ENDIF.
    ENDLOOP.
  ENDIF.

  " --- MFLAG from zidpsd046 (修正：使用 table_line) ---
  IF t_dn[] IS NOT INITIAL.

    REFRESH: lt_loadno, lt_scflag.

    LOOP AT t_dn ASSIGNING <fs_dn>.
      CLEAR ls_loadno.
      ls_loadno-loadno = <fs_dn>-vbeln.
      APPEND ls_loadno TO lt_loadno.
    ENDLOOP.

    SORT lt_loadno BY loadno.
    DELETE ADJACENT DUPLICATES FROM lt_loadno COMPARING loadno.

    IF lt_loadno[] IS NOT INITIAL.

      SELECT loadno
        FROM zidpsd046
        FOR ALL ENTRIES IN @lt_loadno
        WHERE loadno = @lt_loadno-loadno
          AND scflag = 'Y'
        INTO CORRESPONDING FIELDS OF TABLE @lt_scflag.

      SORT lt_scflag BY loadno.

      LOOP AT t_dn ASSIGNING <fs_dn>.
        IF <fs_dn>-docnumber IS NOT INITIAL.

          CLEAR lv_loadno.
          lv_loadno = <fs_dn>-vbeln.

          READ TABLE lt_scflag TRANSPORTING NO FIELDS
            WITH KEY loadno = lv_loadno
            BINARY SEARCH.

          IF sy-subrc = 0.
            <fs_dn>-mflag = 'Y'.
          ELSE.
            <fs_dn>-mflag = 'N'.
          ENDIF.

        ENDIF.
      ENDLOOP.

    ENDIF.

  ENDIF.

  " --- Floor adjustment for IN plants ---
  LOOP AT t_dn ASSIGNING <fs_dn>.
    IF <fs_dn>-werks+0(2) = 'IN'.
      <fs_dn>-floor = <fs_dn>-floor+0(3).
    ENDIF.
  ENDLOOP.

ENDFORM.

*&      Form  check_authorization
*&---------------------------------------------------------------------*
FORM check_authorization.
  CALL FUNCTION 'ZRFC_MM_IDP_0104'
    TABLES
      input  = p_plant[]
      output = p_plant[].

    ENDIF.
    STOP.
  ENDIF.
