FORM check_table_lock
     USING p_matnr p_aufnr p_error.
  DATA: l_gage LIKE seqg3-garg.
  DATA: l_err_num TYPE sy-tabix.
  DATA: l_num TYPE n.
  DATA: l_err_flag(1) TYPE c.
  DATA: l_rsnum LIKE rkpf-rsnum.

  CONCATENATE sy-mandt p_matnr INTO l_gage.

  CLEAR: l_err_num,g_err_str.
  CLEAR: g_enq,g_enq[].
****Check table MARC
  CALL FUNCTION 'ENQUEUE_READ'
    EXPORTING
      gclient = sy-mandt
      gname   = 'MARC'
      garg    = l_gage
    IMPORTING
      number  = l_err_num
    TABLES
      enq     = g_enq.
  IF l_err_num > 0.
    l_num = l_err_num.
    LOOP AT g_enq.
      CONCATENATE 'Material is Locked by:'(050) g_enq-guname
                  '(' g_enq-gtcode ',MARC,' l_num ')'
                  INTO g_err_str.
      EXIT.
    ENDLOOP.
  ENDIF.

  CHECK g_err_str IS INITIAL.

****Check table MARA
  CLEAR: g_enq,g_enq[].
  CALL FUNCTION 'ENQUEUE_READ'
    EXPORTING
      gclient = sy-mandt
      gname   = 'MARA'
      garg    = l_gage
    IMPORTING
      number  = l_err_num
    TABLES
      enq     = g_enq.
  IF l_err_num > 0.
    l_num = l_err_num.
    LOOP AT g_enq.
      CONCATENATE 'Material is Locked by:'(051) g_enq-guname
                  '(' g_enq-gtcode ',MARA,' l_num ')'
                  INTO g_err_str.
      EXIT.
    ENDLOOP.
  ENDIF.

****Check table AUFK(WO header)
  CHECK g_err_str IS INITIAL.
  CLEAR: g_enq,g_enq[].
  CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
    EXPORTING
      input  = p_aufnr
    IMPORTING
      output = p_aufnr.
  CONCATENATE sy-mandt p_aufnr INTO l_gage.
  CALL FUNCTION 'ENQUEUE_READ'
    EXPORTING
      gclient = sy-mandt
      gname   = 'AUFK'
      garg    = l_gage
    IMPORTING
      number  = l_err_num
    TABLES
      enq     = g_enq.
  IF l_err_num > 0.
    l_num = l_err_num.
    LOOP AT g_enq.
      CONCATENATE 'WO is Locked by:'(052) g_enq-guname
                  '(' g_enq-gtcode ',AUFK,' l_num ')'
                  INTO g_err_str.
      EXIT.
    ENDLOOP.
  ENDIF.

****Check table RKPF(WO header)
  CHECK g_err_str IS INITIAL.

  SELECT SINGLE rsnum FROM afko
         INTO l_rsnum
         WHERE aufnr = p_aufnr.
  CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
    EXPORTING
      input  = p_aufnr
    IMPORTING
      output = p_aufnr.
  CONCATENATE sy-mandt l_rsnum INTO l_gage.
  CLEAR: g_enq,g_enq[].
  CALL FUNCTION 'ENQUEUE_READ'
    EXPORTING
      gclient = sy-mandt
      gname   = 'RKPF'
      garg    = l_gage
    IMPORTING
      number  = l_err_num
    TABLES
      enq     = g_enq.
  IF l_err_num > 0.
    l_num = l_err_num.

                  '(' g_enq-gtcode ',RKPF,' l_num ')'

*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*      -->P_AUFK_WERKS  text
*----------------------------------------------------------------------*
FORM get_postdate USING pl_werks
                  CHANGING message.
  DATA : p_lfgja  LIKE marv-lfgja,
         p_lfmon  LIKE marv-lfmon,
         p_lfmon2 LIKE marc-lfmon.

  CALL FUNCTION 'ZRFC_FI_IDP_GET_PERIOD'
    EXPORTING
      i_bwkey = pl_werks
      i_uname = sy-uname
    IMPORTING
      e_monat = p_lfmon.

  g_postdate = sy-datum.
  CLEAR p_lfmon2.
  p_lfgja = g_postdate+0(4).
  IF p_lfmon <> g_postdate+4(2).
    IF g_postdate+4(2) = '12'.
      g_postdate+0(4) = p_lfgja + 1.
      p_lfmon2 = 01.
    ELSE.
      p_lfmon2 = g_postdate+4(2) + 1.
    ENDIF.
    IF p_lfmon2 = p_lfmon.
      g_postdate+4(2) = p_lfmon.
      g_postdate+6(2) = '01'.
    ELSE.
      CONCATENATE 'Please post on '(074) sy-datum+0(4) p_lfmon
'(month)!'(075)
      INTO message.
    ENDIF.
  ENDIF.
ENDFORM.                    " get_postdate

            input  = t_zidppp099-daufnr
          IMPORTING
            output = t_zidppp099-daufnr.
      ELSE.
        t_zidppp099-daufnr = messtab-msgv1.
        t_zidppp099-mess1 = 'Generate Dis Wo Failed!'.
      ENDIF.
      MODIFY t_zidppp099 TRANSPORTING daufnr pflag mess1
              WHERE werks = input-werks
               AND fmatnr = input-a_matnr
               AND ffloor = input-floor
               AND  mess1 = ''.
      CLEAR t_zidppp099.
    ENDIF.
  ENDLOOP.
  MODIFY zidppp099 FROM TABLE t_zidppp099.
ENDFORM.

FORM d_post .
  DATA:l_post_wa TYPE zidppp099.
  DATA:l_post LIKE TABLE OF zidppp099.
  DATA:input_285 LIKE zidpmms007 OCCURS 0 WITH HEADER LINE.
  DATA:outitab_285 LIKE zidpmms008 OCCURS 0 WITH HEADER LINE.
  DATA:l_pt LIKE zidppp099-dflag.
  PERFORM select_screen.

*篩選FG轉GB已開立轉撥單數據
  LOOP AT t_zidppp099 WHERE ( pflag = 'C' OR pflag = 'D' ) AND sflag = 'G'  AND d_doc <> '' AND d_mblnr = '' .
    MOVE-CORRESPONDING t_zidppp099 TO l_post_wa.
    APPEND l_post_wa TO l_POST.
  ENDLOOP.
*檢查單據是否過帳
  SELECT DISTINCT a~werks,a~d_doc,a~pallet,b~docstas,b~mblnr
    FROM @l_post AS a ##ITAB_KEY_IN_SELECT
      LEFT JOIN zidpmmwhh AS b
        ON a~d_doc = b~docnumber
  INTO TABLE @DATA(ll).
  LOOP AT ll ASSIGNING FIELD-SYMBOL(<fsl>).
    CLEAR: input_285,outitab_285,l_pt.
    FREE :input_285,outitab_285.
    IF <fsl>-docstas = '1' .
      MODIFY t_zidppp099[] FROM VALUE #( pflag = 'P' d_mblnr = <fsl>-mblnr  post_mes = '單據已過帳')
         TRANSPORTING pflag d_mblnr post_mes
           WHERE pallet = <fsl>-pallet
             AND faufnr = <fsl>-d_doc.
      CONTINUE.
    ENDIF.
    input_285-document = <fsl>-d_doc.
    input_285-employid = 'F8888888'.
    input_285-werks = <fsl>-werks.
    APPEND input_285.

    CALL FUNCTION 'ZRFC_MM_IDP_0285'
      TABLES
        input   = input_285
        outitab = outitab_285.
    READ TABLE outitab_285 INDEX 1.
    IF  outitab_285-mblnr <> ''.
      l_pt = 'P'.
      COMMIT WORK AND WAIT.
    ENDIF.
    MODIFY t_zidppp099[] FROM VALUE #( d_mblnr = outitab_285-mblnr post_mes = outitab_285-msg pflag = l_pt )
      TRANSPORTING d_mblnr post_mes pflag
        WHERE pallet = <fsl>-pallet
         AND d_doc = <fsl>-d_doc.

  ENDLOOP.
*更新表
  MODIFY zidppp099 FROM TABLE t_zidppp099.

ENDFORM.
